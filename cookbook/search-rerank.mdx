Improve your search quality by combining fast vector search with semantic reranking. This two-stage approach delivers better results than vector search alone.

## Why Reranking?

Vector search is fast but can miss nuanced relevance. Reranking fixes this:

```
Query: "How do I reset my password?"

Vector Search Results (fast, approximate):
1. "Password policies overview" (similar words, wrong topic)
2. "How to reset your password" (correct!)
3. "Password security best practices" (related but wrong)

After Reranking (semantic understanding):
1. "How to reset your password" ← Promoted!
2. "Password recovery options" ← Found in position 8
3. "Forgot password troubleshooting" ← Found in position 12
```

## Prerequisites

- Heroku app with Cohere models (embeddings + rerank)
- Heroku Postgres with pgvector
- Indexed documents (see [RAG Tutorial](/cookbook/complete-rag-tutorial))

## Basic Search + Rerank

```python
import os
from openai import OpenAI
import psycopg2
import requests

# Initialize clients
embedding_client = OpenAI(
    base_url=os.getenv("EMBEDDING_URL") + "/v1",
    api_key=os.getenv("EMBEDDING_KEY")
)

def get_db():
    return psycopg2.connect(os.getenv("DATABASE_URL"))

def search_with_rerank(
    query: str,
    initial_k: int = 20,
    final_k: int = 5
) -> list[dict]:
    """Two-stage search: vector retrieval + semantic reranking."""

    # Stage 1: Fast vector search for candidates
    candidates = vector_search(query, top_k=initial_k)

    if not candidates:
        return []

    # Stage 2: Rerank candidates for relevance
    reranked = rerank_results(query, candidates, top_n=final_k)

    return reranked

def vector_search(query: str, top_k: int = 20) -> list[dict]:
    """Fast approximate search using pgvector."""

    # Generate query embedding
    response = embedding_client.embeddings.create(
        model="cohere-embed-multilingual",
        input=[query],
        extra_body={"input_type": "search_query"}
    )
    query_embedding = response.data[0].embedding

    # Search database
    conn = get_db()
    cur = conn.cursor()

    cur.execute(
        """
        SELECT id, content, metadata,
               1 - (embedding <=> %s::vector) AS similarity
        FROM documents
        ORDER BY embedding <=> %s::vector
        LIMIT %s
        """,
        (query_embedding, query_embedding, top_k)
    )

    results = [
        {
            "id": row[0],
            "content": row[1],
            "metadata": row[2],
            "vector_score": row[3]
        }
        for row in cur.fetchall()
    ]

    cur.close()
    conn.close()

    return results

def rerank_results(
    query: str,
    candidates: list[dict],
    top_n: int = 5
) -> list[dict]:
    """Rerank candidates using Cohere Rerank."""

    # Call rerank API
    response = requests.post(
        f"{os.getenv('INFERENCE_URL')}/v1/rerank",
        headers={
            "Authorization": f"Bearer {os.getenv('INFERENCE_KEY')}",
            "Content-Type": "application/json"
        },
        json={
            "model": "cohere-rerank-3-5",
            "query": query,
            "documents": [c["content"] for c in candidates],
            "top_n": top_n,
            "return_documents": False
        }
    )

    rerank_results = response.json()["results"]

    # Combine scores and reorder
    reranked = []
    for result in rerank_results:
        idx = result["index"]
        candidate = candidates[idx].copy()
        candidate["rerank_score"] = result["relevance_score"]
        candidate["original_position"] = idx
        reranked.append(candidate)

    return reranked

# Example usage
results = search_with_rerank(
    query="How do I deploy a Python app to Heroku?",
    initial_k=20,
    final_k=5
)

for i, result in enumerate(results):
    print(f"{i+1}. Score: {result['rerank_score']:.3f}")
    print(f"   Original position: {result['original_position']}")
    print(f"   Content: {result['content'][:100]}...")
    print()
```

## Production Search Service

```python
from dataclasses import dataclass
from typing import Optional
import time

@dataclass
class SearchResult:
    id: str
    content: str
    score: float
    metadata: dict
    source: str  # "vector" or "reranked"

class SearchService:
    def __init__(self):
        self.embedding_client = OpenAI(
            base_url=os.getenv("EMBEDDING_URL") + "/v1",
            api_key=os.getenv("EMBEDDING_KEY")
        )
        self.rerank_url = f"{os.getenv('INFERENCE_URL')}/v1/rerank"
        self.rerank_key = os.getenv("INFERENCE_KEY")

    def search(
        self,
        query: str,
        top_k: int = 5,
        use_rerank: bool = True,
        filters: Optional[dict] = None
    ) -> tuple[list[SearchResult], dict]:
        """
        Full search with optional reranking and filtering.

        Returns: (results, metrics)
        """
        metrics = {"start_time": time.time()}

        # Vector search
        candidates = self._vector_search(
            query,
            top_k=top_k * 4 if use_rerank else top_k,
            filters=filters
        )
        metrics["vector_time"] = time.time() - metrics["start_time"]
        metrics["vector_candidates"] = len(candidates)

        if not candidates:
            return [], metrics

        # Optional reranking
        if use_rerank and len(candidates) > 1:
            rerank_start = time.time()
            results = self._rerank(query, candidates, top_k)
            metrics["rerank_time"] = time.time() - rerank_start
            metrics["source"] = "reranked"
        else:
            results = candidates[:top_k]
            metrics["source"] = "vector"

        metrics["total_time"] = time.time() - metrics["start_time"]
        metrics["result_count"] = len(results)

        return results, metrics

    def _vector_search(
        self,
        query: str,
        top_k: int,
        filters: Optional[dict] = None
    ) -> list[SearchResult]:
        # Generate embedding
        response = self.embedding_client.embeddings.create(
            model="cohere-embed-multilingual",
            input=[query],
            extra_body={"input_type": "search_query"}
        )
        embedding = response.data[0].embedding

        # Build query with filters
        conn = get_db()
        cur = conn.cursor()

        base_query = """
            SELECT id, content, metadata,
                   1 - (embedding <=> %s::vector) AS similarity
            FROM documents
        """

        params = [embedding, embedding]

        if filters:
            conditions = []
            for key, value in filters.items():
                conditions.append(f"metadata->>'{key}' = %s")
                params.append(value)
            base_query += " WHERE " + " AND ".join(conditions)

        base_query += " ORDER BY embedding <=> %s::vector LIMIT %s"
        params.extend([embedding, top_k])

        cur.execute(base_query, params)

        results = [
            SearchResult(
                id=str(row[0]),
                content=row[1],
                score=row[3],
                metadata=row[2] or {},
                source="vector"
            )
            for row in cur.fetchall()
        ]

        cur.close()
        conn.close()

        return results

    def _rerank(
        self,
        query: str,
        candidates: list[SearchResult],
        top_n: int
    ) -> list[SearchResult]:
        response = requests.post(
            self.rerank_url,
            headers={
                "Authorization": f"Bearer {self.rerank_key}",
                "Content-Type": "application/json"
            },
            json={
                "model": "cohere-rerank-3-5",
                "query": query,
                "documents": [c.content for c in candidates],
                "top_n": top_n,
                "return_documents": False
            }
        )

        rerank_data = response.json()["results"]

        reranked = []
        for item in rerank_data:
            candidate = candidates[item["index"]]
            reranked.append(SearchResult(
                id=candidate.id,
                content=candidate.content,
                score=item["relevance_score"],
                metadata=candidate.metadata,
                source="reranked"
            ))

        return reranked

# Usage
service = SearchService()

results, metrics = service.search(
    query="deploy python heroku",
    top_k=5,
    use_rerank=True,
    filters={"category": "tutorials"}
)

print(f"Found {metrics['result_count']} results in {metrics['total_time']:.3f}s")
print(f"Vector search: {metrics['vector_time']:.3f}s ({metrics['vector_candidates']} candidates)")
if "rerank_time" in metrics:
    print(f"Reranking: {metrics['rerank_time']:.3f}s")
```

## Comparing With vs Without Reranking

```python
def evaluate_search_quality(queries: list[str], ground_truth: dict):
    """Compare search quality with and without reranking."""

    service = SearchService()
    results = {
        "vector_only": {"mrr": 0, "hits_at_5": 0},
        "with_rerank": {"mrr": 0, "hits_at_5": 0}
    }

    for query in queries:
        relevant_ids = ground_truth.get(query, [])

        # Vector only
        vector_results, _ = service.search(query, top_k=5, use_rerank=False)
        vector_ids = [r.id for r in vector_results]

        # With reranking
        rerank_results, _ = service.search(query, top_k=5, use_rerank=True)
        rerank_ids = [r.id for r in rerank_results]

        # Calculate MRR (Mean Reciprocal Rank)
        for i, doc_id in enumerate(vector_ids):
            if doc_id in relevant_ids:
                results["vector_only"]["mrr"] += 1 / (i + 1)
                break

        for i, doc_id in enumerate(rerank_ids):
            if doc_id in relevant_ids:
                results["with_rerank"]["mrr"] += 1 / (i + 1)
                break

        # Calculate Hits@5
        if any(doc_id in relevant_ids for doc_id in vector_ids):
            results["vector_only"]["hits_at_5"] += 1

        if any(doc_id in relevant_ids for doc_id in rerank_ids):
            results["with_rerank"]["hits_at_5"] += 1

    # Average
    n = len(queries)
    for method in results:
        results[method]["mrr"] /= n
        results[method]["hits_at_5"] /= n

    return results

# Example evaluation
test_queries = [
    "How to deploy to Heroku",
    "Database connection errors",
    "Environment variables setup"
]

ground_truth = {
    "How to deploy to Heroku": ["doc_123", "doc_456"],
    "Database connection errors": ["doc_789"],
    "Environment variables setup": ["doc_101", "doc_102"]
}

evaluation = evaluate_search_quality(test_queries, ground_truth)
print(f"Vector Only - MRR: {evaluation['vector_only']['mrr']:.3f}")
print(f"With Rerank - MRR: {evaluation['with_rerank']['mrr']:.3f}")
```

## Cost Considerations

| Operation | Cost | Notes |
|-----------|------|-------|
| Vector search | Embedding cost only | ~$0.10 / 1M tokens |
| Reranking | $2 / 1,000 searches | Each search can rank up to 1,000 docs |

**Optimization tip**: Only rerank when quality matters. For autocomplete or quick suggestions, vector search alone may suffice.
