Build a production-ready RAG (Retrieval-Augmented Generation) application in under 30 minutes. This tutorial takes you from zero to a working knowledge-based chatbot that answers questions using your own documents.

## What You'll Build

A Q&A system that:
1. Loads and chunks your documents
2. Generates embeddings with Cohere
3. Stores vectors in Heroku Postgres with pgvector
4. Retrieves relevant context for user questions
5. Generates accurate, grounded answers with Claude

## Prerequisites

- Heroku account with CLI installed
- Python 3.9+
- A Heroku app with:
  - Managed Inference add-on (Claude model + Cohere embeddings)
  - Heroku Postgres (Essential-0 or higher)

```sql
-- Enable pgvector
CREATE EXTENSION IF NOT EXISTS vector;

-- Create documents table
CREATE TABLE documents (
    id SERIAL PRIMARY KEY,
    content TEXT NOT NULL,
    metadata JSONB DEFAULT '{}',
    embedding VECTOR(1024),  -- Cohere outputs 1024 dimensions
    created_at TIMESTAMP DEFAULT NOW()
);

-- Create index for fast similarity search
CREATE INDEX ON documents USING hnsw (embedding vector_cosine_ops);
```

## Step 3: Create the RAG Application

Create `rag.py`:

```python
import os
import psycopg2
from openai import OpenAI
from dotenv import load_dotenv

load_dotenv()

# Initialize clients
embedding_client = OpenAI(
    base_url=os.getenv("EMBEDDING_URL") + "/v1",
    api_key=os.getenv("EMBEDDING_KEY")
)

chat_client = OpenAI(
    base_url=os.getenv("INFERENCE_URL") + "/v1",
    api_key=os.getenv("INFERENCE_KEY")
)

def get_db_connection():
    return psycopg2.connect(os.getenv("DATABASE_URL"))

# ----- STEP 1: Chunk Documents -----
def chunk_text(text: str, chunk_size: int = 500, overlap: int = 50) -> list[str]:
    """
    Split text into overlapping chunks for embedding.

    Why chunking matters:
    - Embedding models have token limits (Cohere: ~512 tokens optimal)
    - Smaller chunks = more precise retrieval
    - Overlap ensures we don't cut sentences mid-thought

    Args:
        text: The full document text
        chunk_size: Target words per chunk (500 ≈ 375 tokens)
        overlap: Words shared between consecutive chunks

    Returns:
        List of text chunks ready for embedding
    """
    words = text.split()
    chunks = []

    # Slide window across text with overlap
    # Example: chunk_size=500, overlap=50 means we step 450 words each time
    for i in range(0, len(words), chunk_size - overlap):
        chunk = " ".join(words[i:i + chunk_size])
        if chunk.strip():  # Skip empty chunks
            chunks.append(chunk)

    return chunks

# ----- STEP 2: Generate Embeddings -----
def generate_embeddings(texts: list[str]) -> list[list[float]]:
    """
    Generate embeddings for a batch of texts.

    Important considerations:
    - Cohere allows max 96 texts per request
    - Use "search_document" for content being indexed
    - Use "search_query" when embedding user queries (done in retrieve_context)
    - This distinction improves retrieval quality significantly

    Args:
        texts: List of text chunks to embed

    Returns:
        List of embedding vectors (1024 dimensions for Cohere)
    """
    embeddings = []

    # Process in batches of 96 (Cohere's limit)
    for i in range(0, len(texts), 96):
        batch = texts[i:i + 96]

        response = embedding_client.embeddings.create(
            model=os.getenv("EMBEDDING_MODEL_ID"),
            input=batch,
            # IMPORTANT: Use "search_document" for content, "search_query" for queries
            extra_body={"input_type": "search_document"}
        )

        # Extract embedding vectors from response
        embeddings.extend([item.embedding for item in response.data])

        # Log progress for large batches
        if len(texts) > 96:
            print(f"  Embedded {min(i + 96, len(texts))}/{len(texts)} chunks")

    return embeddings

# ----- STEP 3: Store in Database -----
def store_documents(chunks: list[str], metadata: dict = None):
    """Store document chunks with their embeddings."""
    embeddings = generate_embeddings(chunks)

    conn = get_db_connection()
    cur = conn.cursor()

    for chunk, embedding in zip(chunks, embeddings):
        cur.execute(
            """
            INSERT INTO documents (content, metadata, embedding)
            VALUES (%s, %s, %s)
            """,
            (chunk, metadata or {}, embedding)
        )

    conn.commit()
    cur.close()
    conn.close()

    print(f"Stored {len(chunks)} document chunks")

# ----- STEP 4: Retrieve Relevant Context -----
def retrieve_context(query: str, top_k: int = 3) -> list[str]:
    """Find the most relevant document chunks for a query."""
    # Generate query embedding
    response = embedding_client.embeddings.create(
        model=os.getenv("EMBEDDING_MODEL_ID"),
        input=[query],
        extra_body={"input_type": "search_query"}
    )
    query_embedding = response.data[0].embedding

    # Search database
    conn = get_db_connection()
    cur = conn.cursor()

    cur.execute(
        """
        SELECT content, 1 - (embedding <=> %s::vector) AS similarity
        FROM documents
        ORDER BY embedding <=> %s::vector
        LIMIT %s
        """,
        (query_embedding, query_embedding, top_k)
    )

    results = cur.fetchall()
    cur.close()
    conn.close()

    return [row[0] for row in results]

# ----- STEP 5: Generate Answer -----
def answer_question(question: str) -> str:
    """Answer a question using RAG."""
    # Retrieve relevant context
    context_chunks = retrieve_context(question)

    if not context_chunks:
        return "I don't have any relevant information to answer that question."

    # Format context with citations
    context = "\n\n".join([
        f"[Source {i+1}]: {chunk}"
        for i, chunk in enumerate(context_chunks)
    ])

    # Generate answer
    response = chat_client.chat.completions.create(
        model=os.getenv("INFERENCE_MODEL_ID"),
        messages=[
            {
                "role": "system",
                "content": """You are a helpful assistant that answers questions based on the provided context.

Rules:
- Only use information from the provided context
- Cite sources using [Source N] format
- If the context doesn't contain the answer, say "I don't have information about that"
- Be concise and accurate"""
            },
            {
                "role": "user",
                "content": f"Context:\n{context}\n\nQuestion: {question}"
            }
        ],
        temperature=0.2,
        max_tokens=500
    )

    return response.choices[0].message.content

# ----- Main Functions -----
def index_document(filepath: str):
    """Load and index a document."""
    with open(filepath, "r") as f:
        content = f.read()

    chunks = chunk_text(content)
    store_documents(chunks, {"source": filepath})
    print(f"Indexed {filepath}: {len(chunks)} chunks")

def chat():
    """Interactive Q&A loop."""
    print("RAG Chatbot ready! Type 'quit' to exit.\n")

    while True:
        question = input("You: ").strip()
        if question.lower() == "quit":
            break
        if not question:
            continue

        answer = answer_question(question)
        print(f"\nAssistant: {answer}\n")

if __name__ == "__main__":
    import sys

    if len(sys.argv) > 1:
        # Index documents
        for filepath in sys.argv[1:]:
            index_document(filepath)
    else:
        # Start chat
        chat()
```

## Deploy to Heroku

Create `Procfile`:
```
web: python app.py
```

Create `app.py` for a web interface:

```python
from flask import Flask, request, jsonify
from rag import answer_question

app = Flask(__name__)

@app.route("/ask", methods=["POST"])
def ask():
    data = request.json
    question = data.get("question", "")

    if not question:
        return jsonify({"error": "No question provided"}), 400

    answer = answer_question(question)
    return jsonify({"answer": answer})

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=int(os.environ.get("PORT", 5000)))
```

Deploy:
```bash
pip freeze > requirements.txt
git init && git add . && git commit -m "RAG app"

git push heroku main
```

## Optimization Tips

### Improve Retrieval Quality

```python
# Use reranking for better results
def retrieve_with_rerank(query: str, top_k: int = 3) -> list[str]:
    # Get more candidates than needed
    candidates = retrieve_context(query, top_k=10)

    # Rerank with Cohere
    rerank_response = embedding_client.post(
        "/v1/rerank",
        json={
            "model": "cohere-rerank-3-5",
            "query": query,
            "documents": candidates,
            "top_n": top_k
        }
    )

    return [candidates[r["index"]] for r in rerank_response["results"]]
```

### Handle Long Documents

```python
# Use larger chunks with more overlap for technical docs
chunks = chunk_text(content, chunk_size=800, overlap=200)
```

### Add Metadata Filtering

```sql
-- Add filters to your search
SELECT content FROM documents
WHERE metadata->>'category' = 'technical'
ORDER BY embedding <=> %s::vector
LIMIT 5;
```
